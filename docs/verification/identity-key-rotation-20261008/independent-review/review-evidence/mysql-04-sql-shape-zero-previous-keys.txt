namespace Tests\Feature\ProductionIdentity;

use App\Domain\Customers\CustomerIdentityChallenges;
use Illuminate\Support\Facades\Crypt;

use Tests\Support\CustomerIdentityFixtures;
use Tests\Support\ProductionIdentityFixture;
use Tests\TestCase;

/** Review tool (not a product test): captures the MySQL general log for the identity request paths with one key. */
class ReviewSqlShapeCaptureTest extends TestCase
{
    use ProductionIdentityFixture;

    public function test_capture_identity_request_sql_with_a_single_key(): void
    {
        config(['app.key' => 'base64:'.base64_encode(str_repeat('1', 32)), 'app.previous_keys' => []]);
        app()->forgetInstance('encrypter');
        Crypt::clearResolvedInstance('encrypter');
        $this->identitySetup();
        $this->travelTo(now()->startOfSecond());
        $file = getenv('REVIEW_GENERAL_LOG');
        $pdo = DB::connection()->getPdo();
        $pdo->exec("SET GLOBAL general_log_file='".$file."'");
        $pdo->exec('SET GLOBAL general_log=ON');
        try {
            $pdo->query("SELECT 'REVIEW-MARK-START'")->fetchAll();
            $enroll = $this->requestIdentity('enroll', 'buyer@example.test', str_repeat('e', 64));
            $this->requestIdentity('enroll', 'buyer@example.test', str_repeat('e', 64)); // exact replay
            $this->completeIdentity($enroll);
            $this->requestIdentity('recover', 'buyer@example.test', str_repeat('c', 64));
            $this->requestIdentity('enroll', 'other@example.test', str_repeat('d', 64));
            $pdo->query("SELECT 'REVIEW-MARK-LEGACY'")->fetchAll();
            CustomerIdentityFixtures::configure();
            app(CustomerIdentityChallenges::class)->request('enroll', 'legacy@example.test', (string) Str::uuid(), CustomerIdentityFixtures::OWNER);
            $pdo->query("SELECT 'REVIEW-MARK-END'")->fetchAll();
        } finally {
            $pdo->exec('SET GLOBAL general_log=OFF');
        }
        $this->assertSame(1, DB::table('production_identity_origins')->count());
    }
