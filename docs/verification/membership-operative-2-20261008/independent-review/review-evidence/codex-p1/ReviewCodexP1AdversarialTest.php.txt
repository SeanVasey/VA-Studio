    /** Ties: five hints share one received_at second; pages of 2 must examine each exactly once and miss none. */
    public function test_hints_tied_on_received_at_across_page_boundaries_are_each_examined_exactly_once(): void
    {
        F::binding();
        $this->at(5);
        $ids = [];
        foreach (range(1, 5) as $n) {
            $ids[] = 'evt_SYNTHETICTIE'.$n;
            $this->lose('evt_SYNTHETICTIE'.$n, 'in_SYNTHETICTIE'.$n);
        }
        $this->assertSame(1, DB::table('production_membership_billing_events')->distinct()->count('received_at'), 'All five share one second.');
        $this->at(30);
        Queue::fake();

        $seen = [];
        $after = null;
        $pages = 0;
        do {
            $options = ['--limit' => 2] + ($after === null ? [] : ['--after' => $after]);
            $this->assertSame(0, Artisan::call('membership-billing:sweep-hints', $options));
            $output = Artisan::output();
            $pages++;
            preg_match_all('/^uncovered event=([0-9a-f]{12}) /m', $output, $m);
            array_push($seen, ...$m[1]);
            $after = preg_match('/^Next cursor: (\S+)$/m', $output, $c) === 1 ? $c[1] : null;
        } while ($after !== null && $pages < 10);

        $expected = array_map(fn ($id) => substr(BillingValues::hash('event', F::ACCOUNT, 'test', $id), 0, 12), $ids);
        sort($expected);
        $sorted = $seen;
        sort($sorted);
        $this->assertSame($expected, $sorted, 'Every tied hint examined exactly once.');
        $this->assertSame(3, $pages);

        $this->assertSame(0, Artisan::call('membership-billing:sweep-hints', ['--limit' => 1, '--all' => true, '--dispatch' => true]));
        $this->assertStringContainsString('Scan complete: 5 hint(s) examined in 5 page(s).', Artisan::output());
        Queue::assertPushed(RetrieveMembershipInvoice::class, 5);
    }

    /** APP_KEY rotation with the old key in previous_keys: an old cursor fails closed; a fresh pass still reads the ledger. */

    /**
     * Failure model, not a code path: committed position rows are lost and the counter is reissued (asynchronous-replica failover,
     * innodb_flush_log_at_trx_commit/sync_binlog != 1 on an OS crash, or a point-in-time restore) while a retrieval whose provider
     * reads began BEFORE a hint is still in flight with its old position. Positions are bare ids, so that retrieval can append with a
     * reissued id that the new primary handed to a later retrieval, and the stale read then covers the hint and becomes the tail.
     * Simulated by dropping only the positions delete guard, deleting the "lost" rows, rewinding the counter and restoring the guard.
     */
    public function test_probe_after_a_counter_regression_an_in_flight_stale_retrieval_borrows_a_reissued_position(): void
    {
        $binding = F::binding();
        $first = (new BillingReconciliation(new RehearsalBillingGateway(F::graph())))->retrieve($binding['id'], F::INVOICE);
        $this->assertSame(['settled', 1], [$first['outcome'], $first['sequence']]);
        $ledger = new BillingLedger;
        $lostDummy = $ledger->startRetrieval(); // another retrieval begun just before the failure (its row is lost too)
        $hintPayload = $this->event('evt_SYNTHETICREISSUE', F::INVOICE, 'invoice.payment_failed');
        $state = [];

        $this->at(10);
        $stale = new InterleavingBillingGateway(new RehearsalBillingGateway(F::graph()), function () use (&$state, $lostDummy, $hintPayload, $ledger): void {
            // The stale retrieval holds position P (allocated and committed before its reads). Now the database loses P and the dummy.
            $p = (int) DB::table('production_membership_billing_positions')->max('id');
            $state['stale_position'] = $p;
            $this->losePositions([$lostDummy, $p], $lostDummy - 1);
            // A hint arrives at the (new) primary after the stale reads began, and its dispatch is lost.
            $this->at(20);
            $this->lose('evt_SYNTHETICREISSUE', F::INVOICE, 'invoice.payment_failed', $hintPayload);
            $state['hint_position'] = (int) DB::table('production_membership_billing_events')->value('hint_position');
            // The hint's own retrieval begins after it and is allocated the reissued id P.
            $state['fresh_position'] = $ledger->startRetrieval();
            $this->at(30);
        });
        $appended = (new BillingReconciliation($stale))->retrieve($binding['id'], F::INVOICE);

        fwrite(STDERR, "\n[review] driver=".DB::getDriverName().' reissue state='.json_encode($state + ['stale_appended_position' => $appended['retrieval_position'],
            'stale_outcome' => $appended['outcome'], 'stale_sequence' => $appended['sequence']])."\n");
        $this->assertLessThan($state['fresh_position'], $state['hint_position']);
        $this->assertSame($state['stale_position'], $state['fresh_position'], 'The counter reissued the stale retrieval\'s id to a newer retrieval.');
        $this->assertSame(['settled', 2, $state['fresh_position']], [$appended['outcome'], $appended['sequence'], $appended['retrieval_position']],
            'The stale read was admitted under the reissued position and is now the tail.');

        // The hint is now "covered" by a retrieval whose reads began before it arrived.
        Queue::fake();
        $this->at(40);
        $this->assertNull($this->receive($hintPayload)['scheduled'], 'Observed: the stale read covers the hint.');
        Queue::assertNothingPushed();

        // The genuinely newer retrieval can no longer append under its own position.
        try {
            $ledger->append($ledger->existingInvoice($ledger->binding($binding['id'], (new \App\Domain\Memberships\Billing\BillingPolicy)->current()), F::INVOICE),
                new \App\Domain\Memberships\Billing\BillingVerdict('unknown', 'provider_unavailable', ['invoice_ref' => F::INVOICE]),
                self::T0 + 40, 'synthetic_rehearsal', $state['fresh_position'], CarbonImmutable::now('UTC'));
            $this->fail('The reissued position is already used.');
        } catch (Throwable $e) {
            fwrite(STDERR, '[review] fresh append refusal: '.get_class($e).' '.($e instanceof BillingException ? $e->reason : substr($e->getMessage(), 0, 60))."\n");
            $this->assertTrue(true);
        }
    }

    /**

     * refuse it as `schema` (fail closed) and must not adopt it by creating the positions table.
     */
    public function test_a_database_with_the_previous_259000_shape_is_refused_and_not_adopted(): void
    {
        $source = shell_exec('git show 83821fc0:app/Domain/Memberships/Billing/BillingSchema.php 2>/dev/null');
        if (! is_string($source) || ! str_contains($source, 'final class BillingSchema')) {
            $this->markTestSkipped('git history unavailable');
        }
        $class = 'BillingSchemaAt83821fc0';
        if (! class_exists('App\\Domain\\Memberships\\Billing\\'.$class, false)) {
            eval('?>'.str_replace('final class BillingSchema', 'final class '.$class, $source));
        }
        $pdo = DB::connection()->getPdo();
        $driver = DB::getDriverName();
        foreach (array_reverse(BillingSchema::TABLES) as $logical) {
            $pdo->exec('DROP TABLE IF EXISTS '.($driver === 'mysql' ? '`'.$logical.'`' : '"'.$logical.'"'));
        }
        $old = 'App\\Domain\\Memberships\\Billing\\'.$class;
        (new $old)->up();
        (new $old)->assertOwned($pdo);
        $this->assertFalse($this->tableExists('production_membership_billing_positions'));

        foreach (['assertOwned' => fn () => (new BillingSchema)->assertOwned($pdo), 'up' => fn () => (new BillingSchema)->up()] as $label => $operation) {
            try {
                $operation();
                $this->fail($label.' must refuse the previous shape.');
            } catch (BillingException $e) {
                fwrite(STDERR, "\n[review] driver={$driver} previous-shape {$label} refused: {$e->reason}\n");
                $this->assertContains($e->reason, ['schema', 'namespace', 'shadowed_schema']);
            }
        }
        $this->assertFalse($this->tableExists('production_membership_billing_positions'), 'The previous shape is not adopted.');

        // The intake fails closed on it too.
        $this->expectException(BillingException::class);
        $this->receive($this->event('evt_SYNTHETICOLDSHAPE', F::INVOICE, 'invoice.paid'));
    }

    private function tableExists(string $name): bool
    {
        $s = DB::connection()->getPdo()->prepare(DB::getDriverName() === 'sqlite' ? "SELECT 1 FROM main.sqlite_master WHERE type='table' AND name=?"
            : 'SELECT 1 FROM information_schema.TABLES WHERE TABLE_SCHEMA=DATABASE() AND TABLE_NAME=?');
        $s->execute([$name]);

        return $s->fetchColumn() !== false;
    }

    /** Simulates lost committed position rows and a counter rewound to `$counter` (the delete guard is restored byte-for-byte). */
    private function losePositions(array $ids, int $counter): void
    {
        $pdo = DB::connection()->getPdo();
        $driver = DB::getDriverName();
        $schema = new BillingSchema;
        $logical = BillingSchema::TABLES[4];
        $name = DB::connection()->getTablePrefix().$logical;
        $table = $schema->table($logical);
        $guards = (new ReflectionMethod(BillingSchema::class, 'guards'))->invoke($schema, $logical, $driver, $name, $table);
        $guard = $name.'_delete';
        $pdo->exec($driver === 'mysql' ? 'DROP TRIGGER `'.$guard.'`' : 'DROP TRIGGER "'.$guard.'"');
        $pdo->exec('DELETE FROM '.$table.' WHERE id IN ('.implode(',', array_map('intval', $ids)).')');
        if ($driver === 'mysql') {
            // ALTER re-renders existing CHECK literals with `_ascii` introducers, which the owned-schema check refuses (observed), so the
            // simulation re-adds the bounds CHECK through the application connection in the same statement.
            $check = (new ReflectionMethod(BillingSchema::class, 'definition'))->invoke($schema, $logical, $driver)['check'];
            $pdo->exec('ALTER TABLE '.$table.' AUTO_INCREMENT = '.($counter + 1).', DROP CHECK `'.$name.'_bounds`, ADD CONSTRAINT `'.$name.'_bounds` CHECK ('.$check.')');
        } else {
            $pdo->prepare('UPDATE sqlite_sequence SET seq = ? WHERE name = ?')->execute([$counter, $name]);
        }
        $pdo->exec($guards[$guard]['sql']);
        $schema->assertOwned($pdo);
    }

    private function cursor(string $output): string
    {
        $this->assertSame(1, preg_match('/^Next cursor: (\S+)$/m', $output, $match), $output);

        return $match[1];
    }

    private function lose(string $eventId, string $invoiceRef, string $type = 'invoice.paid', ?string $payload = null): void
    {
        try {
            $this->receive($payload ?? $this->event($eventId, $invoiceRef, $type));
        } catch (BindingResolutionException) {
            // Expected: this lane binds no gateway, so the post-commit dispatch fails as a queue outage would.
        }
    }

    private function receive(string $payload): array
    {
        return (new BillingWebhookIntake)->receive($payload, WebhookSignature::generateSignatureHeader($payload, self::SECRET));
    }

    private function event(string $id, string $invoiceRef, string $type): string
    {
        $invoice = [...F::graph()['invoice'], 'id' => $invoiceRef,
            'parent' => ['type' => 'subscription_details', 'subscription_details' => ['subscription' => F::SUBSCRIPTION]]];
        $values = ['id' => $id, 'object' => 'event', 'api_version' => BillingProviderPin::API_VERSION, 'created' => F::PERIOD_START,
