<?php

namespace App\Domain\Grants\Paid;

use RuntimeException;

final class PaidGrantException extends RuntimeException
{
    public function __construct(public readonly int $status = 503)
    {
        parent::__construct('Paid grant request unavailable.');
    }

    public static function require(bool $condition, int $status = 503): void
    {
        if (! $condition) {
            \paidNativeDiagnostic("consumer_refusal", $status, null, debug_backtrace(DEBUG_BACKTRACE_IGNORE_ARGS));
            throw new self($status);
        }
    }
}
