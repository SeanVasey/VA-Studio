function paidNativeDiagnostic(string $stage, int $status, ?string $reason, array $trace): void {

    file_put_contents('/tmp/paid-native-http-diagnostic/refusals.jsonl', json_encode(['at_ns'=>hrtime(true),'stage'=>$stage,'status'=>$status,'reason'=>$reason,'trace'=>$safe], JSON_THROW_ON_ERROR)."\n", FILE_APPEND);
}
