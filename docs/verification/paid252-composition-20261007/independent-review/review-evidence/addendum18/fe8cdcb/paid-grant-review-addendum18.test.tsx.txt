type Route = (init?: RequestInit) => Promise<Response>;
/** Per-path queues; the last answer repeats. Records method and body of every call. */
function server(routes: Record<string, Route[]>) {
  const calls: { path: string; method: string; body: unknown }[] = [];
  vi.spyOn(globalThis, 'fetch').mockImplementation((input: RequestInfo | URL, init?: RequestInit) => {
    const path = String(input); calls.push({ path, method: init?.method ?? 'GET', body: init?.body ? JSON.parse(String(init.body)) : null });
    const queue = routes[path]; if (!queue?.length) throw new Error(`unexpected ${path}`);
    return (queue.length > 1 ? queue.shift()! : queue[0])(init);
  });
  return calls;
}
const ok = (body: unknown): Route => () => Promise.resolve(response(body));

describe('review addendum 18: the refusal render bump', () => {
  /** Counts React commits of the page. */
  function counted() { const commits = { n: 0 }; const view = render(<Profiler id="paid" onRender={() => { commits.n++; }}><PaidGrantJourney /></Profiler>); return { commits, view }; }
  async function submittedDownload() {
    await byReference(orderId); await authorizeKind('master_wav', 1);
    fireEvent.click(button('Download authorized file'));
    return document.querySelector<HTMLIFrameElement>('iframe[title="Paid license attachment response"]')!;
  }

  it('N1: a refusal from a frame removed by a denial renders nothing and resurrects nothing', async () => {
    vi.spyOn(HTMLFormElement.prototype, 'submit').mockImplementation(() => {});
    server({ [finalizePath(orderId)]: [ok({ origin: complete })], [authorizePath(batchId)]: [ok(auth(authX1, 'master_wav'))], '/paid-grants/index': [status(403)] });
    const { commits } = counted();
    const frame = await submittedDownload();
    fireEvent.click(button('Open paid licenses'));
    await screen.findByRole('link', { name: 'Open a fresh sign-in page' });
    expect(frame.isConnected).toBe(false);
    const html = dom(), before = commits.n;
    await act(async () => { refuseFrame(frame); });
    expect(commits.n).toBe(before);
    expect(dom()).toBe(html);
    expect(reopens()).toHaveLength(0);
    expect(query('Retry the authorized download')).not.toBeInTheDocument();
  });

  it('N2: a refusal from a frame removed by pagehide renders nothing and resurrects nothing', async () => {
    vi.spyOn(HTMLFormElement.prototype, 'submit').mockImplementation(() => {});
    server({ [finalizePath(orderId)]: [ok({ origin: complete })], [authorizePath(batchId)]: [ok(auth(authX1, 'master_wav'))] });
    const { commits } = counted();
    const frame = await submittedDownload();
    await act(async () => { window.dispatchEvent(new Event('pagehide')); });
    expect(frame.isConnected).toBe(false);
    const html = dom(), before = commits.n;
    await act(async () => { refuseFrame(frame); });
    expect(commits.n).toBe(before);
    expect(dom()).toBe(html);
    expect(reopens()).toHaveLength(0);
  });

  it('N3: a refusal from a frame removed by unmount renders nothing and reports no error', async () => {
    vi.spyOn(HTMLFormElement.prototype, 'submit').mockImplementation(() => {});
    const errors = vi.spyOn(console, 'error');
    server({ [finalizePath(orderId)]: [ok({ origin: complete })], [authorizePath(batchId)]: [ok(auth(authX1, 'master_wav'))] });
    const { commits, view } = counted();
    const frame = await submittedDownload();
    view.unmount();
    expect(frame.isConnected).toBe(false);
    const before = commits.n;
    await act(async () => { refuseFrame(frame); });
    expect(commits.n).toBe(before);
    expect(errors).not.toHaveBeenCalled();
  });

  it('N4: a non-current refusal during an in-flight read renders once, sends nothing, and the read still commits unaborted', async () => {
    vi.spyOn(HTMLFormElement.prototype, 'submit').mockImplementation(() => {});
    let release!: (r: Response) => void, signal: AbortSignal | undefined;
    const calls = server({ [finalizePath(orderId)]: [ok({ origin: complete })], [authorizePath(batchId)]: [ok(auth(authX1, 'master_wav'))],
      [`/paid-grants/origins/${batchId}/downloads`]: [(init) => { signal = init?.signal ?? undefined; return new Promise<Response>(resolve => { release = resolve; }); }] });
    const { commits } = counted();
    const frame = await submittedDownload();
    // The status read makes the frame non-current and is still in flight when the refusal arrives.
    fireEvent.click(button('Refresh preparation and download status'));
    await waitFor(() => expect(screen.getByLabelText('Paid license journey')).toHaveAttribute('aria-busy', 'true'));
    const sent = calls.length, before = commits.n;
    await act(async () => { refuseFrame(frame); });
    expect(commits.n).toBe(before + 1);
    expect(calls).toHaveLength(sent);
    expect(signal?.aborted).toBe(false);
    expect(screen.getByLabelText('Paid license journey')).toHaveAttribute('aria-busy', 'true');
    expect(screen.queryByText(/download was refused/)).not.toBeInTheDocument();
    expect(screen.getByText(`Order ${orderId}`)).toBeInTheDocument();
    // The read was not abandoned: its answer still commits and offers the retry of the exact unused authorization.
    await act(async () => { release(response(unusedStatus(authX1))); });
    await idle();
    expect(signal?.aborted).toBe(false);
    expect(button('Retry the authorized download')).toBeEnabled();
    expect(dom()).not.toContain(tokenOf(authX1));
  });
});
