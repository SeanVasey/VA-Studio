import sys
path, name = sys.argv[1], sys.argv[2]
src = open(path).read()
M = {
 'M1': ("orderId = p.origin!.orderId;", "orderId = '';"),
 'M2': ("i.originId !== origin?.id && all.findIndex(j => j.originId === i.originId) === n", "i.originId !== origin?.id"),
 'M3': ("i.originId !== origin?.id && all.findIndex", "true && all.findIndex"),
 'M4': ("const unshownIssued = denied ? [] : issued", "const unshownIssued = issued"),
 'M5': ("key={i.originId} disabled={busy}", "key={i.originId} disabled={false}"),
}
old, new = M[name]
assert src.count(old) == 1, (name, src.count(old))
open(path, 'w').write(src.replace(old, new))
print(f"applied {name}: {old!r} -> {new!r}")
