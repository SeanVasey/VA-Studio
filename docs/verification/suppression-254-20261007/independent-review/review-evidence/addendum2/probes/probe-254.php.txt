function conn(string $mode, string $tz): PDO
{
    $pdo = new PDO('mysql:host=127.0.0.1;port=3494;dbname=vaseyaudio_review_supp254d_probe;charset=utf8mb4', 'root', 'review-supp254d-only',

    $pdo->exec('SET SESSION sql_mode='.$pdo->quote($mode));
    $pdo->exec('SET SESSION time_zone='.$pdo->quote($tz));

    return $pdo;
}

$results = [];
function probe(string $group, string $table, array $row, string $modeName, string $mode, string $tz, bool $ignore = false, string $label = ''): array
{
    $pdo = conn($mode, $tz);
    $count = fn () => (int) $pdo->query("SELECT COUNT(*) FROM `$table`")->fetchColumn();
    $pdo->beginTransaction();
    $before = $count();
    $r = ['group' => $group, 'table' => $table, 'label' => $label, 'mode' => $modeName.($ignore ? '+INSERT IGNORE' : ''), 'tz' => $tz,
        'created_at_in' => $row['created_at'], 'public_id_in' => $row['public_id'] ?? null];
    $cols = array_keys($row);
    $sql = 'INSERT '.($ignore ? 'IGNORE ' : '')."INTO `$table` (`".implode('`,`', $cols).'`) VALUES ('.implode(',', array_fill(0, count($cols), '?')).')';
    try {
        $st = $pdo->prepare($sql);
        $st->execute(array_values($row));
        $id = (int) $pdo->lastInsertId();
        $warnings = $pdo->query('SHOW WARNINGS')->fetchAll(PDO::FETCH_NUM);
        $after = $count();
        if ($after === $before) {
            $r['outcome'] = 'no row (statement ok)';
        } else {
            $stored = $pdo->query("SELECT created_at, CAST(created_at AS CHAR) c, UNIX_TIMESTAMP(created_at) u".(isset($row['public_id']) ? ', public_id, HEX(public_id) hx, CHAR_LENGTH(public_id) n' : '')." FROM `$table` WHERE id=$id")->fetch(PDO::FETCH_ASSOC);
            $pdo->exec("SET SESSION time_zone='+00:00'");
            $utc = $pdo->query("SELECT created_at FROM `$table` WHERE id=$id")->fetchColumn();
            $r['outcome'] = 'ACCEPTED';
            $r['stored_session'] = $stored['created_at'];
            $r['stored_utc'] = $utc;
            $r['runtime_shape_ok(utc read)'] = ProductionFeatureShape::timestamp($utc);
            if (isset($row['public_id'])) {
                $r['public_id_stored_hex'] = $stored['hx'];
                $r['public_id_len'] = (int) $stored['n'];
                $r['isUuid(stored)'] = Str::isUuid($stored['public_id']);
            }
        }
        $r['warnings'] = array_map(fn ($w) => $w[1].' '.$w[2], $warnings);
    } catch (PDOException $e) {
        $code = $e->errorInfo[1] ?? null;
        $msg = (string) ($e->errorInfo[2] ?? $e->getMessage());
        $r['outcome'] = str_contains($msg, 'Retained production suppression refused') ? 'refused by GUARD ('.$code.' 45000)' : 'refused by COLUMN/server ('.$code.' '.$msg.')';
        $after = $count();
    }
    $r['rows_left_in_txn'] = $after - $before;
    $pdo->rollBack();

    return $r;
}

function commit(string $table, array $row): int
{
    $pdo = conn(STRICT, '+00:00');
    $cols = array_keys($row);
    $pdo->prepare("INSERT INTO `$table` (`".implode('`,`', $cols).'`) VALUES ('.implode(',', array_fill(0, count($cols), '?')).')')->execute(array_values($row));

    return (int) $pdo->lastInsertId();
}

foreach ($rows as $table => $make) {
    foreach ($malformed as $v) {
        foreach ($modes as $mn => $m) {
            $results[] = probe('a-malformed', $table, $make(['created_at' => $v]), $mn, $m, '+00:00');
        }
        $results[] = probe('a-malformed', $table, $make(['created_at' => $v]), 'strict(laravel)', STRICT, '+00:00', true);
    }
    foreach ($coerced as $v) {
        foreach (['strict(laravel)' => STRICT, 'nonstrict(empty)' => ''] as $mn => $m) {
            $results[] = probe('b-coerced', $table, $make(['created_at' => $v]), $mn, $m, '+00:00');
        }
    }
    if ($table !== 'production_suppression_confirmations') {
        foreach ($uuidShapes as $label => $v) {
            foreach (['strict(laravel)' => STRICT, 'nonstrict(empty)' => ''] as $mn => $m) {
                $results[] = probe('c-public-id', $table, $make(['public_id' => $v]), $mn, $m, '+00:00', false, $label);
            }
        }
    }
    foreach ($dst as [$tz, $v, $label]) {
        foreach (['strict(laravel)' => STRICT, 'nonstrict(empty)' => ''] as $mn => $m) {
            $results[] = probe('dst', $table, $make(['created_at' => $v]), $mn, $m, $tz, false, $label);
        }
    }
    // Positive control in each mode and zone: the exact valid t0 row is accepted (then rolled back).
    foreach (['strict(laravel)' => STRICT, 'nonstrict(empty)' => ''] as $mn => $m) {
        $results[] = probe('control', $table, $make([]), $mn, $m, '+00:00', false, 'valid t0');
    }
    if ($table === 'production_suppression_intents') {

        $results[] = probe('d-intent-before-withdrawal', $table, $make(['withdrawal_event_id' => $p['e2'], 'created_at' => '2026-01-01 00:00:10']), 'strict(laravel)', STRICT, '+00:00', false, 'intent at the same instant (control)');

    }
}
