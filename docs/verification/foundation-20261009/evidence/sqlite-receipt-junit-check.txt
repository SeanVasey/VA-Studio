# Run database-receipts.py's own junit() check on local SQLite shard JUnit, against the shard's listed tests.
import importlib.util, sys
from pathlib import Path
root = Path.cwd()
spec = importlib.util.spec_from_file_location("receipts", root / "scripts/ci/database-receipts.py")
m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m)
skip_pairs = m.sqlite_skip_pairs(root)
for shard, junit_path in zip((1, 2), sys.argv[1:3]):
    expected = m.inventory((root / f"phpunit-ci-local-{shard}-tests.xml").read_bytes(), str(root))
    raw = Path(junit_path).read_bytes()
    try:
        r = m.junit(raw, expected, str(root), "sqlite", skip_pairs)
        print(f"shard {shard}: accepted", {k: r[k] for k in ("reported_cases", "executed_cases", "skipped_cases")})
    except m.ReceiptError as e:
        print(f"shard {shard}: REJECTED: {e}")
