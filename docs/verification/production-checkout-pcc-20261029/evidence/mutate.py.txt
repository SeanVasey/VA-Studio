#!/usr/bin/env python3
"""Apply one textual mutation, run the named tests, restore the file, record killed/survived."""
import subprocess, sys, pathlib
ROOT = pathlib.Path('/home/user/wt/pcc')
PU = '/home/user/notes/pcc-ev/pu.sh'
MUTATIONS = [
    ('M1 gate admits truthy values', 'app/Http/ProductionCheckout/ProductionCheckoutMount.php', "($parent['mounted'] ?? null) === true", "($parent['mounted'] ?? null) == true",
     ['tests/Feature/ProductionCheckoutComposition/ProductionCheckoutMountTest.php']),
    ('M2 gate admits staging', 'app/Http/ProductionCheckout/ProductionCheckoutMount.php', "public const ENVIRONMENTS = ['local', 'testing', 'production'];", "public const ENVIRONMENTS = ['local', 'testing', 'production', 'staging'];",
     ['tests/Feature/ProductionCheckoutComposition/ProductionCheckoutMountTest.php']),
    ('M3 gate ignores the environment', 'app/Http/ProductionCheckout/ProductionCheckoutMount.php', "&& self::environment($app, $instances);", ";",
     ['tests/Feature/ProductionCheckoutComposition/ProductionCheckoutMountTest.php']),
    ('M4 privacy skips the runtime mount check', 'app/Http/Middleware/ProductionCheckoutPrivacy.php', "if (! ProductionCheckoutMount::admitted()) {", "if (false) {",
     ['tests/Feature/ProductionCheckoutComposition/ProductionCheckoutMountTest.php']),
    ('M5 provider registered without the gate', 'app/Providers/AppServiceProvider.php', "if (ProductionCheckoutMount::admitted()) {", "if (true) {",
     ['tests/Feature/ProductionCheckoutComposition/ProductionCheckoutMountTest.php']),
    ('M6 exclusivity authority not consulted', 'app/Domain/Commerce/ProductionCheckout/ExclusiveInventoryGate.php', "        CheckoutException::require(self::storeMaySellExclusives(), 'exclusivity_authority');\n", "",
     ['--filter', 'test_an_exclusive_order|test_an_unreadable|test_detection', 'tests/Feature/ProductionCheckoutComposition/ProductionCheckoutExclusivityTest.php']),
    ('M7 exclusivity authority inverted', 'app/Domain/Commerce/ProductionCheckout/ExclusiveInventoryGate.php', "CheckoutException::require(self::storeMaySellExclusives(), 'exclusivity_authority');", "CheckoutException::require(! self::storeMaySellExclusives(), 'exclusivity_authority');",
     ['--filter', 'test_an_exclusive_order', 'tests/Feature/ProductionCheckoutComposition/ProductionCheckoutExclusivityTest.php']),
    ('M8 ledger failure treated as yes', 'app/Domain/Commerce/ProductionCheckout/ExclusiveInventoryGate.php', "        } catch (Throwable) {\n            return false;", "        } catch (Throwable) {\n            return true;",
     ['--filter', 'test_an_unreadable', 'tests/Feature/ProductionCheckoutComposition/ProductionCheckoutExclusivityTest.php']),
    ('M9 gate not called from the selection', 'app/Domain/Commerce/ProductionCheckout/CurrentSelection.php', "        ExclusiveInventoryGate::refuse($graph, $links);\n", "",
     ['--filter', 'test_an_exclusive_order|test_the_hosted', 'tests/Feature/ProductionCheckoutComposition/ProductionCheckoutExclusivityTest.php']),
    ('M10 command skips the operator switch', 'app/Domain/Commerce/ProductionCheckout/HostedCheckout.php', "        CheckoutException::require(config('production_checkout.operator_reconciliation_enabled') === true\n            && config", "        CheckoutException::require(true\n            && config",
     ['--filter', 'test_it_refuses_before|test_the_domain_operator', 'tests/Feature/ProductionCheckoutComposition/ProductionCheckoutReconcileCommandTest.php']),
]
only = sys.argv[1:]
for name, path, old, new, tests in MUTATIONS:
    if only and name.split()[0] not in only:
        continue
    file = ROOT / path
    source = file.read_text()
    assert source.count(old) == 1, (name, 'anchor count', source.count(old))
    file.write_text(source.replace(old, new))
    try:
        run = subprocess.run([PU, *tests], capture_output=True, text=True, timeout=1500)
        tail = (run.stdout.strip().splitlines() or [''])[-1]
        print(f"{name}: {'KILLED' if run.returncode != 0 else 'SURVIVED'} ({tail})", flush=True)
    finally:
        file.write_text(source)
