<?php

namespace Tests\Feature;

use App\Domain\Contracts\ContractFiles;
use App\Domain\Grants\Free\FreeGrantDocuments;
use App\Domain\Grants\Free\FreeGrantRecords;
use App\Domain\Grants\Free\FreeGrantRendererProcess;
use App\Domain\Grants\ProductionFree\ProductionFreeGrantDocuments;
use App\Domain\Grants\ProductionFree\ProductionFreeGrantRendererProcess;
use App\Domain\Grants\ProductionFree\ProductionFreeGrants;
use Illuminate\Support\Facades\DB;
use Symfony\Component\Process\Process;
use Tests\Support\ProductionFreeGrantFixtures;
use Tests\TestCase;

/**
 * M-16 evidence probe (not a product test): records the exact renderer stdin payload ({input, profile}) and the PDF
 * SHA-256 the CLI produced, so a real PHP-FPM request can render the same payload and compare bytes.
 * Output directory: env M16_CAPTURE_DIR.
 */
final class M16CaptureProductionFreeProbeTest extends TestCase
{
    use ProductionFreeGrantFixtures;

    private function capturing(string $class, string $name): void
    {
        $directory = (string) getenv('M16_CAPTURE_DIR');
        $this->app->instance($class, new $class(function (array $command, string $cwd, array $environment, string $payload) use ($directory, $name): Process {
            file_put_contents($directory.'/'.$name.'.payload.json', $payload);

            return new Process($command, $cwd, $environment, $payload, 60);
        }));
    }

    public function test_capture_production_free(): void
    {
        $this->freeSetup();
        $definition = $this->openDefinition();
        $owner = $this->customer();
        $grants = new ProductionFreeGrants;
        $review = $grants->review($definition['id'], 'Declared Synthetic Buyer', $owner['principal'], $owner['user']);
        $origin = $grants->accept($definition['id'], $this->assentInput($review), $owner['principal'], $owner['user']);
        $this->capturing(ProductionFreeGrantRendererProcess::class, 'production-free');
        (new ProductionFreeGrantDocuments)->render($origin['id']);
        $sha = (string) DB::table('production_free_originals')->value('sha256');
        file_put_contents(getenv('M16_CAPTURE_DIR').'/production-free.cli-sha256', $sha);
        $this->assertMatchesRegularExpression('/\A[a-f0-9]{64}\z/', $sha);
    }
}
